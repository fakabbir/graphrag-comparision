% ── layout-independent float environments ────────────────────────────────
% A float that is too wide for one column must become figure*/table* in the
% two-column build. Declaring the environments here keeps body.tex identical
% for both layouts instead of forking it.
\iftwocol
  \newenvironment{widefig}{\begin{figure*}}{\end{figure*}}
  \newenvironment{widetab}{\begin{table*}}{\end{table*}}
\else
  \newenvironment{widefig}{\begin{figure}}{\end{figure}}
  \newenvironment{widetab}{\begin{table}}{\end{table}}
\fi
